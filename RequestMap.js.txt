export class RequestMap {
  constructor() {
    this.map = new Map();
  }

  set(id, data) {
    this.map.set(id, data);
  }

  get(id) {
    return this.map.get(id);
  }

  has(id) {
    return this.map.has(id);
  }

  delete(id) {
    return this.map.delete(id);
  }

  values() {
    return Array.from(this.map.values());
  }

  clear() {
    this.map.clear();
  }
}
